active xingqiu;
for let i=0; i<6; i=i+1 {
  xingqiu skill;
  xingqiu dash;
  if .xingqiu.burst.ready { xingqiu burst; }
  xingqiu attack:1;
  luminedendro skill;
  if .luminedendro.burst.ready { luminedendro burst; }
  luminedendro attack:1;
  fischl skill;
  fischl attack:1;
  if .ifa.burst.ready { ifa burst; }
  ifa skill;
  while .ifa.nightsoul.points > 10 { ifa attack; }
  luminedendro skill;
  luminedendro attack:2;
  if .fischl.burst.ready { fischl burst; }
  for let r=0; r<3; r=r+1 { ifa attack:3; ifa charge; ifa jump; }
}
