# Iansan's burst buff needs the active character to move (see the Varesa/Iansan file).
let move_speed = 8;
let last_f = 0;
fn mv() {
  movement((f() - last_f) / 60 * move_speed);
  last_f = f();
}

let MAX_NA_COUNT = 18;
fn clorinde_combo() {
  clorinde skill;
  let na_count = 0;
  while .clorinde.status.clorinde-night-watch && na_count < MAX_NA_COUNT {
    if .clorinde.bolratio >= 1 {
      clorinde skill;
    } else {
      clorinde attack;
      na_count = na_count + 1;
    }
  }
  last_f = f();   # no movement counted during her combo
  if .clorinde.burst.ready { clorinde burst; }
}

active mavuika;
# opener
mavuika skill;
iansan skill, attack, burst;
last_f = f();
chevreuse skill[hold=1], burst;
mavuika burst;
mavuika charge, dash; mv();
mavuika charge, dash; mv();
clorinde_combo();
# ongoing
for let i=0; i<5; i=i+1 {
  chevreuse skill[hold=1], attack;
  clorinde attack:3;
  iansan skill, attack, burst;
  last_f = f();
  mavuika skill, burst;
  mavuika charge:2, dash; mv();
  mavuika charge, dash; mv();
  clorinde_combo();
}
