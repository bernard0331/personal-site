#   Valeriy's example build (C6, as Prydwen's teams): Finale of the Deep R5 (craftable), 4pc Noblesse, ATK% main stats; KQM-standard substats from database rotation kqm-532f5039 (same main stats).

#   Cyno's gear for Stellar-Conduct teams (Prydwen: Kitain Cross Spear R5, 4pc Disenchantment in Deep Shadow, ATK%/ATK%/CRIT).
#   Example builds (KQM-standard substats) from KQM Sim Database rotations: yaemiko this spec's gear (see notes), odette kqm-cb9466b0-5d3e-4ed0-87b7-2360bbc64514, valeriy this spec (see notes), cyno this spec's gear (see notes).

cyno add set="disenchantmentindeepshadow" count=4;
cyno add stats hp=4780 atk=311 atk%=0.466 atk%=0.466 cd=0.622;

active yaemiko;
for let i=0; i<6; i=i+1 {
  for let r=0; r<3; r=r+1 { yaemiko skill; }
  for let r=0; r<2; r=r+1 { odette skill; }
  if .valeriy.burst.ready { valeriy burst; }
  valeriy skill;
  valeriy charge;
  cyno skill;
  if .cyno.burst.ready { cyno burst; }
  while .cyno.mods.cyno-q {
    if .cyno.mods.cyno-a1 && .cyno.skill.ready { cyno skill; } else if .cyno.normal == 5 && .stam > 20 { cyno dash; } else { cyno attack; }
  }
}
