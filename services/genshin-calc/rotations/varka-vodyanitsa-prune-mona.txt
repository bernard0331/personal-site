# special_ca = 1 (the owner's choice): the special charged attack (Azure
# Devour) where the database combo uses the special skill (Four Winds'
# Ascension): 3[N1CW] N2sC 3[N1CW] N1sC N1. Both spend the same special
# charges; in the simulator, "charge" during Varka's skill is the special
# charged attack while a charge is available. 0: the combo as the database
# wrote it, 3[N1CW] N2sE 3[N1CW] N1sE N1.
let special_ca = 1;

fn special() {
  if special_ca { varka charge; } else { varka skill; }
}

# combo 2 (the owner's): N1CW N4 sC 3[N1CW] N1sC N1.
# combo 1: the database's 3[N1CW] N2s 3[N1CW] N1s N1.
let combo = 2;

active vodyanitsa;
for let i=0; i<6; i=i+1 {
  vodyanitsa skill;
  prune skill, burst;
  mona skill, burst, attack;
  if combo == 2 {
    varka skill,
          attack, charge, walk,
          attack:4;
  } else {
    varka skill,
          attack, charge, walk,
          attack, charge, walk,
          attack, charge, walk,
          attack:2;
  }
  special();
  varka attack, charge, walk,
        attack, charge, walk,
        attack, charge, walk,
        attack;
  special();
  varka attack;
}
