# Iansan's burst buff needs the active character to move (see the Varesa/Iansan file).
let move_speed = 8;
let last_f = 0;
fn mv() {
  movement((f() - last_f) / 60 * move_speed);
  last_f = f();
}

fn varesa_combo(amt) {
  for let j = 0; j < amt; j = j + 1 {
    if .varesa.skill.ready {
      varesa skill, attack, high_plunge[collision=1];   # ECP
    } else if .xianyun.status.xianyun-airborne-buff {
      varesa jump;
      if .airborne {
        varesa high_plunge[collision=1];                # JP
      }
    } else {
      varesa charge, high_plunge[collision=1];          # CP
    }
    mv();
    if .varesa.energy >= 30 && .varesa.status.apex-drive {
      varesa burst;
    }
  }
}

active furina;
for let i=0; i<6; i=i+1 {
  furina skill, dash, attack, burst;
  iansan skill, attack, burst;
  last_f = f();
  xianyun attack, skill, low_plunge, burst;
  varesa_combo(7);
}
