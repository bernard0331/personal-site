# Team: Mizuki, Faruzan, Odette, Cryo Traveler
# Written for this project on 2026-11-16. Not from the KQM Sim Database.
# Source: KQM's Mizuki quick guide, https://keqingmains.com/q/mizuki-quickguide/
#   Odette 2[E] > Faruzan Q / E C > Traveler E N1C Q > Mizuki E, for her full skill duration.
#   As the guide says, Faruzan bursts when ready and otherwise uses E C, so she bursts every
#   other rotation and rotations are about 17s.
#
# The builds below only make the file runnable on its own (KQM-standard substats from
# the simulator's optimiser). The optimizer replaces them with the player's own.

faruzan char lvl=90/90 cons=6 talent=10,10,10;
faruzan add weapon="elegyfortheend" refine=1 lvl=90/90;
faruzan add set="viridescentvenerer" count=4;
faruzan add stats hp=4780 atk=311 er=0.518 anemo%=0.466 cr=0.311;
faruzan add stats def%=0.062*2 def=19.68*2 hp=253.94*2 hp%=0.0496*2 atk=16.54*2 atk%=0.0496*2 er=0.0551*10 em=19.82*2 cr=0.0331*5 cd=0.0662*11;

luminecryo char lvl=90/90 cons=6 talent=10,10,10;
luminecryo add weapon="exaiphanesblade" refine=5 lvl=90/90;
luminecryo add set="gladiatorsfinale" count=2;

target lvl=100 resist=0.1 radius=2 pos=0,2.4 hp=999999999;
energy every interval=480,720 amount=1;

active odette;
for let i=0; i<6; i=i+1 {
  odette skill, skill;
  if .faruzan.burst.ready { faruzan burst; } else { faruzan skill, aim; }
  if i { luminecryo skill, attack, charge, burst; } else { luminecryo skill; }
  mizuki skill;
  while .mizuki.status.dreamdrifter-state { wait(1); }
}
