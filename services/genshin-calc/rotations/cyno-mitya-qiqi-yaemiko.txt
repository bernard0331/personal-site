#   Example builds (KQM-standard substats) from KQM Sim Database rotations: mitya this spec (see notes), yaemiko this spec's gear (see notes), qiqi kqm-cb9466b0-5d3e-4ed0-87b7-2360bbc64514, cyno this spec's gear (see notes).

cyno add weapon="kitaincrossspear" refine=5 lvl=90/90;
cyno add set="disenchantmentindeepshadow" count=4;
cyno add stats hp=4780 atk=311 atk%=0.466 atk%=0.466 cd=0.622;
cyno add stats def%=0.062*2 def=19.68*2 hp=253.94*2 hp%=0.0496*2 atk=16.54*2 atk%=0.0496*5 er=0.0551*2 em=19.82*2 cr=0.0331*11 cd=0.0662*10;

target lvl=100 resist=0.1 radius=2 pos=0,2.4 hp=999999999;

active mitya;
for let i=0; i<6; i=i+1 {
  mitya skill;
  for let r=0; r<3; r=r+1 { yaemiko skill; }
  qiqi skill;
  qiqi attack:1;
  cyno skill;
  if .cyno.burst.ready { cyno burst; }
  while .cyno.mods.cyno-q {
    if .cyno.mods.cyno-a1 && .cyno.skill.ready { cyno skill; } else if .cyno.normal == 5 && .stam > 20 { cyno dash; } else { cyno attack; }
  }
}
