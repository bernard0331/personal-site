# Sandrone's combo (our sandrone rotations, from KQM's Sandrone guide): 3[C E] (2 at C1+),
# bursting between when ready.
fn sandrone_combo() {
  let n = 3;
  if .sandrone.cons >= 1 { n = 2; }
  for let j=0; j<n; j=j+1 {
    sandrone charge, skill;
    if j < n - 1 && .sandrone.burst.ready { sandrone burst; }
  }
}

active yaemiko;
for let i=0; i<6; i=i+1 {
  for let r=0; r<3; r=r+1 { yaemiko skill; }
  escoffier skill;
  mitya skill;
  sandrone_combo();
}
