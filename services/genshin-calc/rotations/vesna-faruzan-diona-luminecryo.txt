  luminecryo charge;
  if .luminecryo.burst.ready { luminecryo burst; }
  if .faruzan.burst.ready { faruzan burst; }

  vesna attack, charge;
  if .vesna.skill.ready { vesna skill; }
  vesna attack, charge;
  if .vesna.skill.ready { vesna skill; }
  if .vesna.burst.ready { vesna burst; } else { vesna attack:3, charge; }

  vesna attack, charge;
  if .vesna.skill.ready { vesna skill; }
  vesna attack, charge;
  if .vesna.skill.ready { vesna skill; }
