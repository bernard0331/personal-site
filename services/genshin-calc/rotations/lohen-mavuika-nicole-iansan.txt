# Iansan's burst buff needs the active character's movement reported (see the
# Varesa/Iansan rotation): mv() reports move_speed metres per second of field time.
let move_speed = 8;
let last_f = 0;
fn mv() {
  movement((f() - last_f) / 60 * move_speed);
  last_f = f();
}

active nicole;
for let i=0; i<6; i=i+1 {
  nicole skill;
  if .nicole.burst.ready { nicole burst; }
  mavuika skill;
  iansan skill;
  iansan charge;
  if .iansan.burst.ready { iansan burst; }
  last_f = f();
  lohen skill;
  for let r=0; r<2; r=r+1 { lohen attack, charge; }
  if .lohen.burst.ready { lohen burst; } else { lohen skill; }
  for let r=0; r<4; r=r+1 { lohen attack, charge; }
  lohen skill;
  for let r=0; r<5; r=r+1 { lohen attack, charge; }
  lohen skill;
  if .mavuika.burst.ready { mavuika burst; }
}
