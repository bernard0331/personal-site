#   "Dash twice to refresh Iansan's Nightsoul points, then E sQ immediately." The guide says
#   Iansan holds 4pc Noblesse in Flins teams (for his Lunar-Charged damage), so her example
#   build here does. In the simulator Flins's burst during his special state is his special
#   burst (sQ), as in the database's Flins rotations (e.g. Gjb9dqBQkNhp).
#   The guide's two special bursts per rotation need C1's energy; each sQ here happens only
#   when ready, so C0 Flins skips one (with two per rotation at C0 the rotation stalled for
#   minutes waiting on energy). At C1 nothing changes.
#   Notation (the guide's): E skill, C charged attack, D dash, Nx x normal attacks,
#   Q burst, sQ special burst.

# Iansan's burst buff needs the active character to move (see the Varesa/Iansan file).
let move_speed = 8;
let last_f = 0;
fn mv() {
  movement((f() - last_f) / 60 * move_speed);
  last_f = f();
}

active aino;
for let i=0; i<6; i=i+1 {
  aino skill, burst;
  iansan skill, attack, burst;
  last_f = f();
  lynette skill;
  flins skill, attack, skill;
  if .flins.burst.ready { flins burst; }
  flins dash; mv();
  flins attack:3, dash; mv();
  flins attack:4, dash; mv();
  flins dash, dash; mv();
  flins skill;
  if .flins.burst.ready { flins burst; }
  flins attack:4, dash; mv();
  flins attack:5; mv();
}
