active furina;
for let i=0; i<6; i=i+1 {
  furina skill;
  furina dash;
  if .furina.burst.ready { furina burst; }
  if .bennett.burst.ready { bennett burst; }
  bennett skill;
  faruzan skill;
  if .faruzan.burst.ready { faruzan burst; }
  ifa skill;
  while .ifa.nightsoul.points > 10 { ifa charge; }
  if .ifa.burst.ready { ifa burst; }
  bennett skill;
