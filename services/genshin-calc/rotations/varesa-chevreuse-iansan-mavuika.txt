# Team: Varesa, Chevreuse, Iansan, Mavuika
# Written for this project on 2026-11-16. Not from the KQM Sim Database.
# Source: KQM's Varesa quick guide, https://keqingmains.com/q/varesa-quickguide/
#   Team "Varesa — Chevreuse — Mavuika — Ororon / Iansan", sample rotation (Iansan):
#     (Varesa ECP) > Chevreuse Q hE > Iansan E (C) Q > Mavuika EQ 3[C2D] > Varesa 2[ECP (Q) ECP]
#   (Varesa ECP) is first rotation only. Iansan's optional (C) is left out. Varesa's (Q) is her
#   short burst, used in that slot when she has the energy and is in the state that allows it
#   (Fiery Passion, or after any plunge at C2+; the simulator tracks this as "apex-drive").
#   At C2+ she also short-bursts after the second ECP of each pair, per the guide's combo notes
#   ("After C2, Varesa can use a short Burst after each Plunge"): about 7% higher for the
#   owner's C2 Varesa than bursting only in the team rotation's (Q) slot.
#   Notation (the guide's): E skill, hE hold skill, C charged attack (for Mavuika on her
#   Flamestrider: Cyclic Charged Attack), C2 two of them, D dash (Mavuika: Sprint Attack),
#   P plunge, Q burst.
#   Plunges hit with their collision damage, as in the database's Varesa rotations.
#   One change from the guide: Chevreuse bursts when ready, (Q) hE, as in the guide's other
#   Chevreuse rotations, instead of every rotation. Bursting every rotation waited on energy
#   for about 8s over 6 rotations with KQM-standard substats and about 100s with the owner's
#   gear (Favonius Lance, HP%/HP%/HP%).
#
# The builds below only make the file runnable on its own (main stats from the guide's
# recommendations, KQM-standard substats). The optimizer replaces them with the player's own.

options swap_delay=12 iteration=1000;

varesa char lvl=90/90 cons=2 talent=10,10,10;
varesa add weapon="vividnotions" refine=1 lvl=90/90;
varesa add set="longnightsoath" count=4;
varesa add stats hp=4780 atk=311 atk%=0.466 electro%=0.466 cr=0.311;
varesa add stats def%=0.062*2 def=19.68*2 hp=253.94*2 hp%=0.0496*2 atk=16.54*2 atk%=0.0496*2 er=0.0551*4 em=19.82*2 cr=0.0331*8 cd=0.0662*10;

chevreuse char lvl=90/90 cons=6 talent=10,10,10;
chevreuse add weapon="favoniuslance" refine=5 lvl=90/90;
chevreuse add set="noblesseoblige" count=4;
chevreuse add stats hp=4780 atk=311 hp%=0.466 hp%=0.466 hp%=0.466;
chevreuse add stats def%=0.062*2 def=19.68*2 hp=253.94*2 hp%=0.0496*6 atk=16.54*2 atk%=0.0496*2 er=0.0551*8 em=19.82*2 cr=0.0331*4 cd=0.0662*2;

iansan char lvl=90/90 cons=6 talent=10,10,10;
iansan add weapon="favoniuslance" refine=5 lvl=90/90;
iansan add set="scrolloftheheroofcindercity" count=4;
iansan add stats hp=4780 atk=311 atk%=0.466 atk%=0.466 atk%=0.466;
iansan add stats def%=0.062*2 def=19.68*2 hp=253.94*2 hp%=0.0496*2 atk=16.54*2 atk%=0.0496*6 er=0.0551*8 em=19.82*2 cr=0.0331*2 cd=0.0662*2;

mavuika char lvl=90/90 cons=1 talent=10,10,10;
mavuika add weapon="serpentspine" refine=5 lvl=90/90;
mavuika add set="obsidiancodex" count=4;
mavuika add stats hp=4780 atk=311 atk%=0.466 pyro%=0.466 cd=0.622;
mavuika add stats def%=0.062*2 def=19.68*2 hp=253.94*2 hp%=0.0496*2 atk=16.54*2 atk%=0.0496*4 er=0.0551*2 em=19.82*2 cr=0.0331*10 cd=0.0662*8;

target lvl=100 resist=0.1 radius=2 pos=0,2.4 hp=999999999;
energy every interval=480,720 amount=1;

fn ecp() {
  varesa skill, attack, high_plunge[collision=1];
}

active varesa;
ecp();                                   # first rotation only
for let i=0; i<6; i=i+1 {
  if .chevreuse.burst.ready { chevreuse burst; }
  chevreuse skill[hold=1];
  iansan skill, burst;
  mavuika skill, burst, charge:2, dash, charge:2, dash, charge:2, dash;
  for let j=0; j<2; j=j+1 {
    ecp();
    if .varesa.energy >= 30 && .varesa.status.apex-drive { varesa burst; }
    ecp();
    if .varesa.cons >= 2 && .varesa.energy >= 30 && .varesa.status.apex-drive { varesa burst; }
  }
}
