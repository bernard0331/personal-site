active columbina;
for let i=0; i<6; i=i+1 {
  columbina skill;
  if .columbina.burst.ready { columbina burst; }
  gorou skill;
  if .gorou.burst.ready { gorou burst; }
  linnea skill;
  if .linnea.burst.ready { linnea burst; }
  aratakiitto attack;
  if .aratakiitto.burst.ready { aratakiitto burst; }
  aratakiitto skill[travel=0];
  while !.aratakiitto.skill.ready { if .aratakiitto.tags.strStack > 0 { aratakiitto charge; } else { aratakiitto attack; } }
  aratakiitto skill[travel=0];
  while .aratakiitto.mods.itto-q { if .aratakiitto.tags.strStack > 0 { aratakiitto charge; } else { aratakiitto attack; } }
}
