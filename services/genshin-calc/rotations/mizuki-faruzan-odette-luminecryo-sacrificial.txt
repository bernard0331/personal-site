# Witch's Revelation is armed by a Swirl, which also starts its 2.5s cooldown, and fires on the
# next skill hit. The cooldown's start is used as the signal: a big hit follows within about
# 75 frames (the next hit fires within 44 frames and travels for 30).

fn first_cast() {
  mizuki skill;
  while !.mizuki.status.mizuki-revelation-icd && .mizuki.status.dreamdrifter-state { wait(1); }
  wait(80); # the big hit fires and lands; Sacrificial gets its chances on the hits so far
}

fn main_cast(useburst) {
  while !.mizuki.skill.ready { wait(1); } # Sacrificial did not proc
  mizuki skill;
  let arms = 0;
  let expiries = 0;
  let prev = 0;
  let bursted = 0;
  while .mizuki.status.dreamdrifter-state {
    let on = .mizuki.status.mizuki-revelation-icd > 0;
    if on && !prev { arms = arms + 1; }
    if prev && !on { expiries = expiries + 1; }
    prev = on;
    if arms >= 4 {
      wait(80); # the fourth big hit fires and lands
      break;
    }
    # Her burst hits 93 frames after the cast: cast it so the hit lands just after the cooldown
    # that follows the third big hit ends.
    if useburst && !bursted && arms == 3 && on && .mizuki.status.mizuki-revelation-icd <= 92 && .mizuki.burst.ready {
      mizuki burst;
      bursted = 1;
    } else {
      wait(1);
    }
  }
}

active faruzan;
for let i=0; i<6; i=i+1 {
  if .faruzan.burst.ready { faruzan burst; } else { faruzan skill, aim; }
  odette skill, skill;
  first_cast();
  if i { luminecryo skill, attack, charge, burst; } else { luminecryo skill, attack, charge; }
  main_cast(0);
}
