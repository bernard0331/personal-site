# Iansan's burst buff is 27% of her ATK at 42+ Nightsoul points and less below that. Her
# points drain by 6 per second, and she regains them from the active character's movement,
# which the simulator only counts when the script reports it (her burst also ends if she runs
# out). mv() reports movement at move_speed metres per second of field time since the last
# call. 8 m/s is the least that keeps her at 42+ through the burst, as the guides assume
# ("easily maintain the maximum buff"); faster movement overflows her points, which at C6 adds
# her damage bonus (about +2% team DPS here at 12 m/s). Without movement her buff fell to
# under 4% of her ATK.
let move_speed = 8;
let last_f = 0;
fn mv() {
  movement((f() - last_f) / 60 * move_speed);
  last_f = f();
}

fn dc() {   # D*: dash that cancels a charged attack, as mav_dc_cancel(6) in the database rotations
  while !.mavuika.dash.ready { wait(1); }
  if .mavuika.nightsoul.points > 16 || .mavuika.status.mavuika-burst {
    mavuika dash[charge_cancel=1, travel=6], charge;
  } else {
    mavuika jump;
  }
}

active mavuika;
for let i=0; i<6; i=i+1 {
  mavuika skill[hold=1], dash;
  xilonen skill;
  if .xilonen.burst.ready { xilonen burst; }
  xilonen attack:2;
  pick_up_crystallize("pyro");   # the shard triggers Archaic Petra, as in kNgftcWMgtfT
  iansan skill, attack;
  if .iansan.burst.ready { iansan burst; } else { delay(30); }
  last_f = f();
  citlali skill;
  if .citlali.burst.ready { citlali burst; }
  citlali attack;
  mavuika swap;
  # recast E (hE C1) while still in Nightsoul; the database script has no skill.ready check,
  # but here her skill can come off cooldown first, and a recast is then not allowed
  if .mavuika.nightsoul.points && !.mavuika.skill.ready { mavuika skill[recast=1]; mavuika charge; }
  while !.mavuika.burst.ready { mavuika attack; }
  mavuika burst; mv();
  mavuika charge, charge[final=1]; mv();
  dc(); mavuika charge[final=1]; mv();
  dc(); mavuika charge[final=1]; mv();
  dc(); mavuika charge[final=1]; mv();
  dc(); mavuika charge[hold=40], charge[final=1]; mv();
}
