#     Nahida E Q > Yae 3[E] > Fischl E > Ifa combo > Yae Q 3[N1 E] > Fischl N2 Q > Ifa 3[N3CJ]
#   Combo: the guide's standard Q E N13 (P): burst when ready, skill, then held Tonicshots (charged attacks in the simulator) while his Nightsoul lasts, as KQM Sim Database Ifa rotations do; the optional plunge is left out.
#   Supports burst when ready (the guide assumes enough ER to burst every rotation).

nahida char lvl=90/90 cons=6 talent=10,10,10;
nahida add weapon="nightweaverslookingglass" refine=5 lvl=90/90;

fischl add stats hp=4780 atk=311 atk%=0.466 electro%=0.466 cr=0.311;

ifa char lvl=90/90 cons=6 talent=9,9,9;
ifa add weapon="evenstar" refine=1 lvl=90/90;

ifa add stats hp=4780 atk=311 em=187*3;

target lvl=100 resist=0.1 radius=2 pos=0,2.4 hp=999999999;
energy every interval=480,720 amount=1;

active nahida;
for let i=0; i<6; i=i+1 {
  nahida skill;
  if .nahida.burst.ready { nahida burst; }
  for let r=0; r<3; r=r+1 { yaemiko skill; }
  fischl skill;
  if .ifa.burst.ready { ifa burst; }
  ifa skill;
  while .ifa.nightsoul.points > 10 { ifa charge; }
  if .yaemiko.burst.ready { yaemiko burst; }
  for let r=0; r<3; r=r+1 { yaemiko attack:1; yaemiko skill; }
  fischl attack:2;
  if .fischl.burst.ready { fischl burst; }
  for let r=0; r<3; r=r+1 { ifa attack:3; ifa charge; ifa jump; }
}
