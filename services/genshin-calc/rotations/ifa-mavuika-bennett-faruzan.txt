#   Team "Ifa — Mavuika — Bennett / Iansan — C6 Faruzan", sample rotation (Bennett):
#     Faruzan E Q > Bennett Q E > Mavuika Q E > Ifa combo > Bennett E
#   Combo: the guide's E N13 Q (P), best for Anemo DPS: burst last, after held Tonicshots.

target lvl=100 resist=0.1 radius=2 pos=0,2.4 hp=999999999;
energy every interval=480,720 amount=1;

active faruzan;
for let i=0; i<6; i=i+1 {
  faruzan skill;
  if .faruzan.burst.ready { faruzan burst; }
  if .bennett.burst.ready { bennett burst; }
  bennett skill;
  if .mavuika.burst.ready { mavuika burst; }
  mavuika skill;
  ifa skill;
  while .ifa.nightsoul.points > 10 { ifa charge; }
  if .ifa.burst.ready { ifa burst; }
  bennett skill;
}
