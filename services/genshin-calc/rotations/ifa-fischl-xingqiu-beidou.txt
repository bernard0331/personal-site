active xingqiu;
for let i=0; i<6; i=i+1 {
  xingqiu skill;
  xingqiu dash;
  if .xingqiu.burst.ready { xingqiu burst; }
  xingqiu attack:1;
  beidou skill;
  beidou attack:1;
  if .beidou.burst.ready { beidou burst; }
  fischl attack:2;
  if .fischl.burst.ready { fischl burst; }
  ifa attack:1;
  if .ifa.burst.ready { ifa burst; }
  ifa skill;
  while .ifa.nightsoul.points > 10 { ifa attack; }
  beidou skill;
  fischl skill;
  ifa attack:3;
  ifa dash;
  ifa attack:3;
}
