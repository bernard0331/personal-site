# Iansan's burst buff needs the active character to move (see the Varesa/Iansan file).
let move_speed = 8;
let last_f = 0;
fn mv() {
  movement((f() - last_f) / 60 * move_speed);
  last_f = f();
}

fn plunge() {
  gaming skill, high_plunge;
  mv();
}

active bennett;
for let i=0; i<6; i=i+1 {
  bennett skill;
  iansan skill, attack, burst;
  last_f = f();
  bennett burst, skill;
  citlali skill, burst;
  gaming burst;
  for let j=0; j<5; j=j+1 {
    sleep(45);   # catch Man Chai during his descent
    plunge();
  }
  sleep(30);
  plunge();
  citlali attack:2;   # refresh Cryo for the last plunge
  plunge();
}
