# Zibai's combo. C1+: EE 4[N4E]. C0: E 4[N4E]. Either way, N1 Q replaces an N4 when Q is ready.
fn zibai_combo() {
  zibai skill;
  if .zibai.cons >= 1 { zibai skill; }
  for let j=0; j<4; j=j+1 {
    if .zibai.burst.ready { zibai attack, burst; } else { zibai attack:4; }
    zibai skill;
  }
}

active columbina;
for let i=0; i<6; i=i+1 {
  if .columbina.burst.ready {
    columbina skill, burst;     # Long Rot
  } else {
    columbina skill;            # Short Rot
  }
  linnea attack, skill;
  illuga burst, skill;
  zibai_combo();
}
