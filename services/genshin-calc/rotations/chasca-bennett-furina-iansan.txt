# Iansan's burst buff needs the active character to move (see the Varesa/Iansan file).
let move_speed = 8;
let last_f = 0;
fn mv() {
  movement((f() - last_f) / 60 * move_speed);
  last_f = f();
}

active furina;
for let i=0; i<6; i=i+1 {
  furina skill, dash;
  if .furina.burst.ready { furina burst; }
  bennett skill, burst, attack;
  iansan skill, attack, burst;
  last_f = f();
  chasca skill;
  for let k=0; k<4; k=k+1 { chasca aim; mv(); }
  if .chasca.burst.ready && .chasca.energy == .chasca.energymax {
    chasca burst;
  } else {
    chasca aim[bullets=4]; mv();
  }
  if .bennett.skill.ready { bennett skill; }
}
wait(82);   # the last 4-bullet shot lands, as in the database rotation
