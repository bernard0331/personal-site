# Iansan's burst buff needs the active character to move (see the Varesa/Iansan file for
# details); mv() reports movement at move_speed metres per second of field time. Kinich's
# grapples cover plenty of ground, as the guide says.
let move_speed = 8;
let last_f = 0;
fn mv() {
  movement((f() - last_f) / 60 * move_speed);
  last_f = f();
}

# Kinich's combo, E 5[N2 shE] attacking clockwise, from KQM Sim Database rotation
# bwDcdkBN8LL7 (Kinich Emilie Mavuika Bennett), with a movement report after each cycle.
fn kinich_combo() {
  kinich skill;
  for let c=0; c<5; c=c+1 {
    kinich attack[direction=1]:2;
    while .kinich.nightsoul.points < 20 && .kinich.nightsoul.state {
      wait(1);
    }
    if .kinich.nightsoul.state {
      kinich skill[hold=1];
    }
    mv();
  }
}

active mavuika;
for let i=0; i<6; i=i+1 {
  if .kinich.burst.ready { kinich burst; }
  mavuika burst, skill;
  bennett burst, skill;
  iansan skill, attack, burst;
  last_f = f();
  kinich_combo();
}
