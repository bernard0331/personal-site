# Zibai's combo (our zibai-illuga-linnea-columbina rotation, from KQM's Zibai guide):
# C1+: EE 4[N4E]. C0: E 4[N4E]. Either way, N1 Q replaces an N4 when Q is ready.
# Spirit Steed's Stride needs 70 Phase Shift Radiance; in teams that trigger
# Lunar-Crystallize less often she attacks (up to 4 more hits) until she has it.
fn zibai_combo() {
  zibai skill;
  if .zibai.cons >= 1 && .zibai.radiance >= 70 { zibai skill; }
  for let j=0; j<4; j=j+1 {
    if .zibai.burst.ready { zibai attack, burst; } else { zibai attack:4; }
    let k = 0;
    while .zibai.status.zibai-skill > 0 && .zibai.radiance < 70 && k < 4 { zibai attack; k = k + 1; }
    if .zibai.radiance >= 70 { zibai skill; }
  }
}

active zhongli;
for let i=0; i<6; i=i+1 {
  zhongli skill[hold=1];
  zhongli dash;
  aino skill;
  aino attack:1;
  illuga skill;
  if .illuga.burst.ready { illuga burst; }
  if .aino.burst.ready { aino burst; }
  aino attack:1;
  zibai_combo();
}
